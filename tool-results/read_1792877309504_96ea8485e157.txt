    60→/** FNV-1a — deterministik: angka/bundle konsisten per layanan (stabil antar build). */
    61→function hashU32(seed: string): number {
    62→  let h = 2166136261;
    63→  for (let i = 0; i < seed.length; i++) {
    64→    h ^= seed.charCodeAt(i);
    65→    h = Math.imul(h, 16777619);
    66→  }
    67→  return h >>> 0;
    68→}

    70→/** 2 layanan pelengkap dari kategori BERBEDA — deterministik per kode. */
    71→function pickBundle(code: string, categoryCode: string, isV3: boolean): CatalogDetailService[] {
    72→  const pool: CatalogDetailService[] = (isV3 ? KATALOG_V3 : KATALOG_V2).filter(
    73→    (x) => x.categoryCode !== categoryCode
    74→  );
    75→  if (pool.length === 0) return [];
    76→  const h = hashU32(code);
    77→  const i1 = h % pool.length;
    78→  let i2 = (i1 + 1 + ((h >>> 8) % Math.max(1, pool.length - 1))) % pool.length;
    79→  if (i2 === i1) i2 = (i1 + 1) % pool.length;
    80→  return [pool[i1], pool[i2]];
    81→}
    82→
    83→/** Klaim permintaan bulanan kategori — deterministik per kode (konsisten, bukan acak tiap load). */
    84→function monthlyDemand(code: string): number {
    85→  return 8 + (hashU32(code) % 14);
    86→}
