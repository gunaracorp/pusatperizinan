    68→function slugify(raw: string): string {
    69→  return raw
    70→    .toLowerCase()
    71→    .trim()
    72→    .replace(/[^a-z0-9\s-]/g, "")
    73→    .replace(/[\s_]+/g, "-")
    74→    .replace(/-+/g, "-")
    75→    .replace(/^-|-$/g, "")
    76→    .slice(0, 90);
    77→}

    91→function postToForm(p: CmsPost): PostForm {
    92→  return {
    93→    id: p.id, slug: p.slug, title: p.title, metaTitle: p.metaTitle,
    94→    metaDescription: p.metaDescription, focusKeyword: p.focusKeyword,
    95→    keywords: p.keywords, category: p.category, excerpt: p.excerpt,
    96→    author: p.author, authorRole: p.authorRole,
    97→    sections: p.sections.length > 0
    98→      ? p.sections.map((s) => ({
    99→          heading: s.heading,
   100→          paragraphs: s.paragraphs.join("\n"),
   101→          bullets: (s.bullets ?? []).join("\n"),
   102→        }))
   103→      : [{ heading: "", paragraphs: "", bullets: "" }],
   104→    faq: p.faq.length > 0 ? p.faq.map((f) => ({ question: f.question, answer: f.answer })) : [{ question: "", answer: "" }],
   105→    ctaText: p.cta?.text ?? "Konsultasi Sekarang",
   106→    ctaAnchor: p.cta?.anchor ?? "#konsultasi",
   107→    relatedArticles: p.relatedArticles.join(", "),
   108→    relatedGuides: p.relatedGuides.join(", "),
   109→    status: p.status, noindex: p.noindex,
   110→  };
   111→}

   113→function formToPayload(f: PostForm) {
   114→  return {
   115→    slug: f.slug || f.title,
   116→    title: f.title,
   117→    metaTitle: f.metaTitle,
   118→    metaDescription: f.metaDescription,
   119→    focusKeyword: f.focusKeyword,
   120→    keywords: f.keywords,
   121→    category: f.category,
   122→    excerpt: f.excerpt,
   123→    author: f.author,
   124→    authorRole: f.authorRole,
   125→    sections: f.sections
   126→      .filter((s) => s.heading.trim())
   127→      .map((s) => ({
   128→        heading: s.heading.trim(),
   129→        paragraphs: s.paragraphs.split("\n").map((x) => x.trim()).filter(Boolean),
   130→        bullets: s.bullets.split("\n").map((x) => x.trim()).filter(Boolean),
   131→      })),
   132→    faq: f.faq
   133→      .filter((q) => q.question.trim() && q.answer.trim())
   134→      .map((q) => ({ question: q.question.trim(), answer: q.answer.trim() })),
   135→    ctaText: f.ctaText,
   136→    ctaAnchor: f.ctaAnchor,
   137→    relatedArticles: f.relatedArticles.split(",").map((x) => x.trim()).filter(Boolean),
   138→    relatedGuides: f.relatedGuides.split(",").map((x) => x.trim()).filter(Boolean),
   139→    status: f.status,
   140→    noindex: f.noindex,
   141→  };
   142→}

   144→/** Skor SEO live — 10 pemeriksaan ala Yoast, bobot total 100 */
   145→function computeSeoAudit(f: PostForm): { score: number; items: SeoItem[] } {
   146→  const kw = f.focusKeyword.trim().toLowerCase();
   147→  const titleSeo = (f.metaTitle || f.title).trim();
   148→  const desc = (f.metaDescription || f.excerpt).trim();
   149→  const slug = slugify(f.slug || f.title);
   150→  const firstPara = f.sections.find((s) => s.heading.trim())?.paragraphs.split("\n").map((x) => x.trim()).filter(Boolean)[0] ?? "";
   151→  const contentSections = f.sections.filter(
   152→    (s) => s.heading.trim() && (s.paragraphs.trim() || s.bullets.trim())
   153→  ).length;
   154→  const realFaq = f.faq.filter((q) => q.question.trim() && q.answer.trim()).length;
   155→
   156→  const items: SeoItem[] = [
   157→    {
   158→      label: "Kata kunci fokus ditentukan", ok: kw.length > 0,
   159→      hint: "Tentukan 1 frasa utama yang dicari calon klien, mis. \u201cbiaya pendirian PT\u201d",
   160→    },
   161→    {
   162→      label: "Kata kunci ada di judul SEO", ok: kw.length > 0 && titleSeo.toLowerCase().includes(kw),
   163→      hint: "Sisipkan kata kunci secara natural di judul",
   164→    },
   165→    {
   166→      label: `Panjang judul SEO 30\u201360 karakter (kini ${titleSeo.length})`,
   167→      ok: titleSeo.length >= 30 && titleSeo.length <= 60,
   168→      hint: "Judul terlalu pendek terpotong peluang; terlalu panjang terpotong di hasil pencarian",
   169→    },
   170→    {
   171→      label: "Kata kunci ada di meta description", ok: kw.length > 0 && desc.toLowerCase().includes(kw),
   172→      hint: "Meta description yang mengandung kata kunci ditebalkan Google",
   173→    },
   174→    {
   175→      label: `Panjang meta description 110\u2013160 karakter (kini ${desc.length})`,
   176→      ok: desc.length >= 110 && desc.length <= 160,
   177→      hint: "Terlalu pendek = informasi kurang; terlalu panjang = terpotong",
   178→    },
   179→    {
   180→      label: "Kata kunci ada di URL (slug)", ok: kw.length > 0 && slug.includes(slugify(kw)),
   181→      hint: "URL yang mengandung kata kunci memperkuat relevansi",
   182→    },
   183→    {
   184→      label: "Kata kunci ada di paragraf pembuka", ok: kw.length > 0 && firstPara.toLowerCase().includes(kw),
   185→      hint: "Sebut kata kunci di 100 kata pertama",
   186→    },
   187→    {
   188→      label: `Minimal 3 bagian berisi (kini ${contentSections})`,
   189→      ok: contentSections >= 3,
   190→      hint: "Artikel mendalam butuh struktur H2 yang jelas",
   191→    },
   192→    {
   193→      label: `Minimal 3 FAQ (kini ${realFaq})`,
   194→      ok: realFaq >= 3,
   195→      hint: "FAQ membuka peluang rich snippet di Google",
   196→    },
   197→    {
   198→      label: `Ringkasan \u2265 100 karakter (kini ${f.excerpt.trim().length})`,
   199→      ok: f.excerpt.trim().length >= 100,
   200→      hint: "Ringkasan tampil di kartu blog & meta default",
   201→    },
   202→  ];
   203→
   204→  const weights = [10, 12, 12, 10, 12, 10, 10, 10, 8, 6];
   205→  const score = items.reduce((n, it, i) => n + (it.ok ? weights[i] : 0), 0);
   206→  return { score, items };
   207→}

   260→    if (!form) return;

   286→      } else {
   287→        toast({ title: "Gagal menyimpan", description: json.error });

   301→        toast({ title: "Artikel dihapus", description: p.slug });
   302→        if (form?.id === p.id) setForm(null);
   303→        await load();
   304→      } else {
   305→        toast({ title: "Gagal menghapus", description: json.error });
   306→      }
   307→    } finally {
   308→      setBusy(false);
   309→    }
   310→  }
