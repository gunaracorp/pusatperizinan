    77→export async function generateMetadata({
    78→  params,
    79→}: {
    80→  params: Promise<{ slug: string[] }>;
    81→}): Promise<Metadata> {
    82→  const { slug } = await params;
    83→  const page = getAnyPage(slug.join("/"));
    84→  // Katalog Premium v2 (PP-XXX): slug di luar dataset lama → coba dataset v2
    85→  const v2 = page ? undefined : resolveKatalogV2(slug.join("/"));
    86→  if (v2) {
    87→    const title = `${v2.name} — Harga ${new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 }).format(v2.discountPrice)} (${v2.code})`;
    88→    const description = `Pengurusan ${v2.name} (${v2.code}) mulai ${new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 }).format(v2.discountPrice)} — estimasi ${v2.estimate}, diterbitkan ${v2.authority}. Syarat lengkap, garansi 100%, pendampingan hingga terbit oleh PusatPerizinan.com.`;
    89→    return {
    90→      title,
    91→      description,
    92→      alternates: langAlternates(`/layanan/${v2.slug}`),
    93→      openGraph: { title, description, url: `/layanan/${v2.slug}`, type: "article", siteName: "PusatPerizinan.com", locale: "id_ID" },
    94→    };
    95→  }
    96→  // Katalog Pro v3 (PP3-XXX): 500 layanan baru — metadata dengan keywords
    97→  const v3 = page ? undefined : resolveKatalogV3(slug.join("/"));
    98→  if (v3) {
    99→    const fmtIdr = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });
   100→    const title = `${v3.name} — Harga ${fmtIdr.format(v3.discountPrice)} (${v3.code})`;
   101→    const description = `Jasa pengurusan ${v3.name} (${v3.code}) mulai ${fmtIdr.format(v3.discountPrice)} — estimasi ${v3.estimate}, diterbitkan ${v3.authority}, masa berlaku ${v3.validity}. Syarat lengkap, garansi 100%, pendampingan hingga terbit oleh PusatPerizinan.com.`;
   102→    return {
   103→      title,
   104→      description,
   105→      keywords: [
   106→        v3.name.toLowerCase(),
   107→        `biaya ${v3.name.toLowerCase()}`,
   108→        `syarat ${v3.name.toLowerCase()}`,
   109→        v3.code,
   110→        v3.authority.toLowerCase(),
   111→        `jasa pengurusan ${v3.authority.toLowerCase()}`,
   112→      ],
   113→      alternates: langAlternates(`/layanan/${v3.slug}`),
   114→      openGraph: { title, description, url: `/layanan/${v3.slug}`, type: "article", siteName: "PusatPerizinan.com", locale: "id_ID" },
   115→      twitter: { card: "summary_large_image", title, description },
   116→    };
   117→  }
   118→
   119→  if (!page) return {};
   120→
   121→  // M05: meta geo dinamis — provinsi/kota diekstrak dari slug halaman
   122→  // (base/region/city/hub wilayah); halaman tanpa lokasi → geo undefined.
   123→  const geo = geoMetaForSlug(page.slug);
   124→
   125→  return {
   126→    title: page.title,
   127→    description: page.metaDesc,
   128→    keywords: page.keywords,
   129→    // M04: canonical absolut + hreflang id-ID/x-default
   130→    alternates: langAlternates(`/layanan/${page.slug}`),
   131→    ...(geo ? { other: geo } : {}),
   132→    openGraph: {
   133→      title: page.title,
   134→      description: page.metaDesc,
   135→      url: `/layanan/${page.slug}`,
   136→      type: "article",
   137→      siteName: "PusatPerizinan.com",
   138→      locale: "id_ID",
   139→    },
   140→    twitter: {
   141→      card: "summary_large_image",
   142→      title: page.title,
   143→      description: page.metaDesc,
   144→    },
   145→    // P0-03: halaman tipis (tier D/E) dinilai noindex,follow
   146→    robots: { index: classifyPage(pageLikeOf(page)).index, follow: true },
   147→  };
   148→}
