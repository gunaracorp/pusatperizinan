    75→export async function generateMetadata({
    76→  params,
    77→}: {
    78→  params: Promise<{ slug: string[] }>;
    79→}): Promise<Metadata> {
    80→  const { slug } = await params;
    81→  const page = getAnyPage(slug.join("/"));
    82→  // Katalog Premium v2 (PP-XXX): slug di luar dataset lama → coba dataset v2
    83→  const v2 = page ? undefined : resolveKatalogV2(slug.join("/"));
    84→  if (v2) {
    85→    const title = `${v2.name} — Harga ${new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 }).format(v2.discountPrice)} (${v2.code})`;
    86→    const description = `Pengurusan ${v2.name} (${v2.code}) mulai ${new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 }).format(v2.discountPrice)} — estimasi ${v2.estimate}, diterbitkan ${v2.authority}. Syarat lengkap, garansi 100%, pendampingan hingga terbit oleh PusatPerizinan.com.`;
    87→    return {
    88→      title,
    89→      description,
    90→      alternates: langAlternates(`/layanan/${v2.slug}`),
    91→      openGraph: { title, description, url: `/layanan/${v2.slug}`, type: "article", siteName: "PusatPerizinan.com", locale: "id_ID" },
    92→    };
    93→  }
    94→
    95→  if (!page) return {};
    96→
    97→  // M05: meta geo dinamis — provinsi/kota diekstrak dari slug halaman
    98→  // (base/region/city/hub wilayah); halaman tanpa lokasi → geo undefined.
    99→  const geo = geoMetaForSlug(page.slug);
   100→
   101→  return {
   102→    title: page.title,
   103→    description: page.metaDesc,
   104→    keywords: page.keywords,
   105→    // M04: canonical absolut + hreflang id-ID/x-default
   106→    alternates: langAlternates(`/layanan/${page.slug}`),
   107→    ...(geo ? { other: geo } : {}),
   108→    openGraph: {
   109→      title: page.title,
   110→      description: page.metaDesc,
   111→      url: `/layanan/${page.slug}`,
   112→      type: "article",
   113→      siteName: "PusatPerizinan.com",
   114→      locale: "id_ID",
   115→    },
   116→    twitter: {
   117→      card: "summary_large_image",
   118→      title: page.title,
   119→      description: page.metaDesc,
   120→    },
   121→    // P0-03: halaman tipis (tier D/E) dinilai noindex,follow
   122→    robots: { index: classifyPage(pageLikeOf(page)).index, follow: true },
   123→  };
   124→}
